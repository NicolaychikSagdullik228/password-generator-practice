## 3. Реализация ключевых функций
### 1. Криптографически стойкая генерация паролей
В отличие от стандартного модуля random, который использует псевдослучайные алгоритмы, в модуле generator.py применен модуль secrets. Он генерирует случайные числа на основе системного энтропийного пула ОС, что делает пароли устойчивыми к криптоанализу.

def generate_password(length: int = 12, use_upper: bool = True, use_digits: bool = True, use_special: bool = True) -> str:
    alphabet = string.ascii_lowercase
    if use_upper:
        alphabet += string.ascii_uppercase
    if use_digits:
        alphabet += string.digits
    if use_special:
        alphabet += string.punctuation

    if not alphabet:
        raise ValueError("Должен быть выбран хотя бы один набор символов.")

    return ''.join(secrets.choice(alphabet) for _ in range(length))

2. Безопасность Мастер-пароля (Хэширование)
###Хранение паролей в открытом виде — критическая уязвимость. Программа реализует концепцию Master Password. Сам пароль нигде не сохраняется — в базу данных (master) записывается исключительно его необратимый криптографический хэш SHA-256. При авторизации введенная строка хэшируется заново и сравнивается со значением из базы данных.

def hash_password(password: str) -> str:
    return hashlib.sha256(password.encode('utf-8')).hexdigest()

def verify_master_password(password: str) -> bool:
    with get_connection() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT password_hash FROM master LIMIT 1")
        result = cursor.fetchone()
        if not result:
            return False
        return result[0] == hash_password(password)
